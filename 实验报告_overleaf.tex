\section{实验目的}
\begin{enumerate}
  \item 在 MNIST 上实现并训练 \textbf{MLP} 与 \textbf{CNN}，比较二者在相同优化设定下的验证集与官方测试集表现。
  \item 在固定协议下开展 \textbf{Part C}：恒定学习率基线、余弦退火学习率、Dropout 正则，分别在 MLP 与 CNN 上观察趋势。
  \item 熟悉训练—验证—测试流程、checkpoint 选择与指标记录方式。
\end{enumerate}

\section{数据与划分}
\begin{itemize}
  \item \textbf{训练图像：}\texttt{train-images-idx3-ubyte.gz}，标签 \texttt{train-labels-idx1-ubyte.gz}。
  \item \textbf{官方测试集：}\texttt{t10k-images-idx3-ubyte.gz} / \texttt{t10k-labels-idx1-ubyte.gz}（仅最终评估；脚本通过 \texttt{-{}-eval\_test}，用 \textbf{best dev} 权重评测）。
  \item \textbf{划分：}打乱后 60k 中取前 10000 为 \textbf{dev}，后 50000 训练；由种子与 \texttt{idx.pickle} 固定。Part B/C 使用 \texttt{-{}-reuse\_idx} 与 Part A 同划分。
\end{itemize}

\section{模型与训练设置概要}

\subsection{MLP（Part A）}
\begin{itemize}
  \item 结构（\texttt{test\_train.build\_model}）：\textbf{784 $\to$ 600 $\to$ 10}，隐藏层 ReLU，Softmax+交叉熵（\texttt{MultiCrossEntropyLoss}）。
  \item 优化：SGD，$lr=0.06$，无调度，无 Dropout（Part A）。
\end{itemize}

\subsection{CNN（Part B）}
\begin{itemize}
  \item 结构见 \texttt{mynn/models.py} 中 \texttt{Model\_CNN}。
  \item 优化设定与 Part A 对齐（公平对比），仅模型不同。
\end{itemize}

\subsection{监控与保存}
\begin{itemize}
  \item \texttt{eval\_interval=100}：约每 100 个 training iteration 在 dev 上评估一次（见 \texttt{mynn/runner.py}）。
  \item 以 dev 准确率最高保存 \texttt{best\_model.pickle}。
  \item \texttt{-{}-save\_fig} 保存曲线 PNG；\texttt{-{}-save\_metrics} 保存 \texttt{.npz}（含 \texttt{best\_score}；若开 \texttt{-{}-eval\_test} 则含 \texttt{test\_acc}/\texttt{test\_loss}）。
\end{itemize}
